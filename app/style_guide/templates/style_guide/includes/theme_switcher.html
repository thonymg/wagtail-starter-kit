{# Vue island — ThemeSwitcher backed by uiStore (Pinia) #}
Theme Switcher — Vue Island + Pinia 

{# stores/ui.js #}
export const useUiStore = defineStore('ui', {
  state: () => ({ theme: localStorage.getItem('picoPreferredColorScheme') ?? 'auto' }),
  actions: {
    setTheme(theme) {
      this.theme = theme
      const resolved = theme === 'auto'
        ? (window.matchMedia('(prefers-color-scheme: dark)').matches ? 'dark' : 'light')
        : theme
      document.documentElement.setAttribute('data-theme', resolved)
      localStorage.setItem('picoPreferredColorScheme', theme)
    },
  },
})

{# islands/ThemeSwitcher.vue #}
<script setup>
import { useUiStore } from '../stores/ui.js'
const ui = useUiStore()
</script>

<template>
  <details class="dropdown">
    <summary role="button" class="contrast">Theme</summary>
    <ul>
      <li v-for="opt in ['auto','light','dark']" :key="opt">
        <a href="#" @click.prevent="ui.setTheme(opt)">{{ opt }}</a>
      </li>
    </ul>
  </details>
</template>
